\section{Purpose and conventions}

We consider one $\phi^4$ field and a marginal probability $p_t$ with $t\in[0,1]$. The construction follows the PIK framework. The direction of the forward process is
\begin{equation}
 p_{\mathrm{tar}}=p_0
 \longrightarrow
 p_1=p_{\mathrm{B}},
\end{equation}
where $p_{\mathrm{tar}}$ is the interacting target distribution and $p_{\mathrm{B}}$ is a simple base distribution. Reverse-time evolution generates target samples from the base distribution.

The unnormalized action is denoted by $\hat S_t$. The partition function, normalized action, and probability density are

\begin{align}
 Z_t
 &=
 \int_{-\infty}^{\infty}\dd\phi\,e^{-\hat S_t(\phi)},
 \\
 S_t(\phi)
 &=
 \hat S_t(\phi)+\log Z_t,
 \\
 p_t(\phi)
 &=
 e^{-S_t(\phi)}
 =
 \frac{e^{-\hat S_t(\phi)}}{Z_t}.
\end{align}

This notation follows the convention of the original PIK papers: the hat always denotes the unnormalized action, and $S_t$ is the normalized action, i.e.\ the negative log likelihood.

The PIK velocity $v_t$ is defined by the deterministic characteristic equation

\begin{equation}
 \frac{\dd\phi_t}{\dd t}
 =
 v_t(\phi_t).
\end{equation}

We adopt the continuity equation

\begin{equation}
 \partial_t p_t(\phi)
 =
 -\partial_\phi\bigl(p_t(\phi)v_t(\phi)\bigr).
 \label{eq:continuity}
\end{equation}

This equation fixes the sign of $v_t$ unambiguously. A kernel defined through a positive total derivative convention has the opposite sign.

The score is

\begin{equation}
 s_t(\phi)
 =
 \partial_\phi\log p_t(\phi)
 =
 -\partial_\phi\hat S_t(\phi).
 \label{eq:score}
\end{equation}

The score is known once the action path is prescribed. It does not by itself determine the probability transport between different values of $t$. The missing information is carried by $v_t$, or equivalently by the forward stochastic drift introduced below.

\section{Target distribution, base distribution, and action path}

We choose the unnormalized target action

\begin{equation}
 \hat S_{\mathrm{tar}}(\phi)
 =
 -\phi^2+\frac{1}{8}\phi^4
 \label{eq:target_action}
\end{equation}

and the standard Gaussian base action

\begin{equation}
 \hat S_{\mathrm{B}}(\phi)
 =
 \frac{1}{2}\phi^2.
 \label{eq:base_action}
\end{equation}

The target has two minima at $\phi=\pm2$. The base density is

\begin{equation}
 p_{\mathrm{B}}(\phi)
 =
 \frac{1}{\sqrt{2\pi}}e^{-\phi^2/2}.
\end{equation}

We use the smoothstep interpolation function

\begin{align}
 \alpha(t)
 &=
 3t^2-2t^3,
 \\
 \dot\alpha(t)
 &=
 6t(1-t).
 \label{eq:schedule}
\end{align}

It satisfies $\alpha(0)=0$, $\alpha(1)=1$, and $\dot\alpha(0)=\dot\alpha(1)=0$, so the
transport velocity vanishes at both endpoints and the dynamics there reduces to
ordinary Langevin diffusion for the corresponding endpoint density.

The action path is

\begin{equation}
 \hat S_t(\phi)
 =
 \bigl(1-\alpha(t)\bigr)\hat S_{\mathrm{tar}}(\phi)
 +
 \alpha(t)\hat S_{\mathrm{B}}(\phi).
 \label{eq:action_path}
\end{equation}

Its explicit form is

\begin{equation}
 \hat S_t(\phi)
 =
 \left(-1+\frac{3}{2}\alpha(t)\right)\phi^2
 +
 \frac{1-\alpha(t)}{8}\phi^4.
 \label{eq:action_explicit}
\end{equation}

The field derivative is

\begin{equation}
 \partial_\phi\hat S_t(\phi)
 =
 \bigl(-2+3\alpha(t)\bigr)\phi
 +
 \frac{1-\alpha(t)}{2}\phi^3.
 \label{eq:action_force}
\end{equation}

The score follows from Eq.~\eqref{eq:score}

\begin{equation}
 s_t(\phi)
 =
 \bigl(2-3\alpha(t)\bigr)\phi
 -
 \frac{1-\alpha(t)}{2}\phi^3.
 \label{eq:score_explicit}
\end{equation}

The explicit time derivative at fixed $\phi$ is

\begin{equation}
 \partial_t\hat S_t(\phi)
 =
 \dot\alpha(t)
 \left(
 \frac{3}{2}\phi^2
 -
 \frac{1}{8}\phi^4
 \right).
 \label{eq:action_time_derivative}
\end{equation}

The derivative of the normalization is

\begin{equation}
 \partial_t\log Z_t
 =
 -\left\langle\partial_t\hat S_t\right\rangle_{p_t}.
 \label{eq:normalization_derivative}
\end{equation}

Consequently,

\begin{equation}
 \partial_tS_t(\phi)
 =
 \partial_t\hat S_t(\phi)
 -
 \left\langle\partial_t\hat S_t\right\rangle_{p_t}.
 \label{eq:normalized_action_time_derivative}
\end{equation}

The path is invariant under $\phi\mapsto-\phi$. The score is odd, the exact PIK velocity is odd, and all densities are even.

\section{PIK transport equation}

Substitution of $p_t=e^{-S_t}$ into Eq.~\eqref{eq:continuity} gives

\begin{align}
 -p_t\partial_tS_t
 &=
 -\partial_\phi\bigl(p_tv_t\bigr),
 \\
 &=
 -p_t\partial_\phi v_t
 +
 p_tv_t\partial_\phi S_t.
\end{align}

After division by $p_t$, the normalized-action form is

\begin{equation}
 \partial_\phi v_t(\phi)
 -
 v_t(\phi)\partial_\phi S_t(\phi)
 =
 \partial_tS_t(\phi).
 \label{eq:pik_normalized}
\end{equation}

Since $\partial_\phi S_t=\partial_\phi\hat S_t$, the unnormalized-action form is

\begin{equation}
 \partial_\phi v_t(\phi)
 -
 v_t(\phi)\partial_\phi\hat S_t(\phi)
 =
 \partial_t\hat S_t(\phi)
 -
 \left\langle\partial_t\hat S_t\right\rangle_{p_t}.
 \label{eq:pik_strong}
\end{equation}

The zero-flux boundary condition is

\begin{equation}
 \lim_{\phi\to\pm\infty}
 p_t(\phi)v_t(\phi)
 =
 0.
 \label{eq:zero_flux}
\end{equation}

Multiplication of Eq.~\eqref{eq:pik_strong} by $p_t$ gives

\begin{equation}
 \partial_\phi\bigl(p_t(\phi)v_t(\phi)\bigr)
 =
 p_t(\phi)
 \left(
 \partial_t\hat S_t(\phi)
 -
 \left\langle\partial_t\hat S_t\right\rangle_{p_t}
 \right).
\end{equation}

The exact one-dimensional solution is therefore

\begin{equation}
 v_t^{\mathrm{exact}}(\phi)
 =
 \frac{1}{p_t(\phi)}
 \int_{-\infty}^{\phi}\dd\varphi\,
 p_t(\varphi)
 \bigl(
 \partial_t\hat S_t(\varphi)
 -
 \langle\partial_t\hat S_t\rangle_{p_t}
 \bigr).
 \label{eq:exact_velocity_left}
\end{equation}

For positive $\phi$, a numerically more stable expression is

\begin{equation}
 v_t^{\mathrm{exact}}(\phi)
 =
 -\frac{1}{p_t(\phi)}
 \int_{\phi}^{\infty}\dd\varphi\,
 p_t(\varphi)
 \bigl(
 \partial_t\hat S_t(\varphi)
 -
 \langle\partial_t\hat S_t\rangle_{p_t}
 \bigr).
 \label{eq:exact_velocity_right}
\end{equation}

Equations~\eqref{eq:exact_velocity_left} and \eqref{eq:exact_velocity_right} are used only as exact references. The weak training method derived below does not use them.

\subsection{Cumulative distribution and exact transport map}

Define the cumulative distribution

\begin{equation}
 C_t(\phi)
 =
 \int_{-\infty}^{\phi}\dd\varphi\,p_t(\varphi).
 \label{eq:cdf}
\end{equation}

Integration of Eq.~\eqref{eq:continuity} gives

\begin{equation}
 \partial_t C_t(\phi)
 =
 -p_t(\phi)v_t(\phi).
\end{equation}

Along a characteristic trajectory,

\begin{equation}
 \frac{\dd}{\dd t}C_t(\phi_t)
 =
 \partial_tC_t(\phi_t)
 +
 p_t(\phi_t)v_t(\phi_t)
 =
 0.
\end{equation}

Hence

\begin{equation}
 C_t(\phi_t)
 =
 C_0(\phi_0).
 \label{eq:cdf_conservation}
\end{equation}

The exact deterministic map is

\begin{equation}
 \phi_t
 =
 C_t^{-1}\bigl(C_0(\phi_0)\bigr).
 \label{eq:quantile_map}
\end{equation}

At $t=1$, $C_1$ is the standard Gaussian cumulative distribution. Equation~\eqref{eq:quantile_map} provides an independent reference for the probability-flow ODE.

\section{A PIK-compatible family of forward stochastic processes}

We fix a nonnegative diffusion coefficient $D(t)$ that depends only on time. The forward stochastic differential equation is

\begin{equation}
 \dd\phi_t
 =
 b_t(\phi_t)\dd t
 +
 \sqrt{2D(t)}\dd W_t.
 \label{eq:forward_sde}
\end{equation}

The associated Fokker--Planck equation is

\begin{equation}
 \partial_t p_t
 =
 -\partial_\phi\bigl(b_tp_t\bigr)
 +
 D(t)\partial_\phi^2p_t.
 \label{eq:fokker_planck}
\end{equation}

Using $\partial_\phi p_t=p_ts_t$, Eq.~\eqref{eq:fokker_planck} becomes

\begin{equation}
 \partial_t p_t
 =
 -\partial_\phi
 \left(
 p_t\bigl(b_t-D(t)s_t\bigr)
 \right).
\end{equation}

Comparison with Eq.~\eqref{eq:continuity} gives

\begin{equation}
 v_t(\phi)
 =
 b_t(\phi)-D(t)s_t(\phi).
 \label{eq:velocity_drift_relation}
\end{equation}

Equivalently,

\begin{equation}
 b_t(\phi)
 =
 D(t)s_t(\phi)+v_t(\phi)
 =
 v_t(\phi)-D(t)\partial_\phi\hat S_t(\phi).
 \label{eq:drift_decomposition}
\end{equation}

We parameterize only the transport velocity by a neural network $v_\theta$ and
reconstruct the forward drift analytically through the same decomposition,

\begin{equation}
 b_\theta(\phi,t)
 =
 v_\theta(\phi,t)-D(t)\partial_\phi\hat S_t(\phi),
 \label{eq:network_drift_decomposition}
\end{equation}

so the analytically known score is never learned; only the probability transport
$v_\theta$ is.

A given PIK path therefore defines a family of forward SDEs. The family is indexed by the chosen diffusion coefficient $D(t)$. The deterministic probability-flow ODE corresponds to $D(t)=0$.

For the first benchmark we take

\begin{equation}
 D(t)=1.
 \label{eq:constant_diffusion}
\end{equation}

The forward SDE then reads

\begin{equation}
 \dd\phi_t
 =
 \bigl(s_t(\phi_t)+v_t(\phi_t)\bigr)\dd t
 +
 \sqrt{2}\dd W_t.
 \label{eq:forward_sde_unit_diffusion}
\end{equation}

At both endpoints, $\dot\alpha=0$ and $v_t=0$. The endpoint dynamics reduces to ordinary Langevin dynamics for the corresponding endpoint density.

\subsection{Reverse-time stochastic equation}

Introduce the increasing reverse time $\tau=1-t$. The reverse-time SDE associated with Eq.~\eqref{eq:forward_sde} is

\begin{align}
 \dd\phi_\tau
 ={}&
 \left(
 -b_{1-\tau}(\phi_\tau)
 +
 2D(1-\tau)s_{1-\tau}(\phi_\tau)
 \right)\dd\tau
 \nonumber\\
 &+
 \sqrt{2D(1-\tau)}\dd\overline W_\tau.
 \label{eq:reverse_sde_general}
\end{align}

Using Eq.~\eqref{eq:drift_decomposition}, the reverse drift becomes

\begin{equation}
 -b_t+2D(t)s_t
 =
 -v_t+D(t)s_t.
 \label{eq:reverse_drift_decomposition}
\end{equation}

For $D=1$,

\begin{equation}
 \dd\phi_\tau
 =
 \bigl(s_{1-\tau}(\phi_\tau)-v_{1-\tau}(\phi_\tau)\bigr)\dd\tau
 +
 \sqrt{2}\dd\overline W_\tau.
 \label{eq:reverse_sde_unit_diffusion}
\end{equation}

The score in this equation is known analytically from Eq.~\eqref{eq:score_explicit}. Only the transport component $v_t$, or equivalently the forward drift $b_t$, has to be learned.

The reverse probability-flow ODE is

\begin{equation}
 \frac{\dd\phi_\tau}{\dd\tau}
 =
 -v_{1-\tau}(\phi_\tau).
 \label{eq:reverse_probability_flow}
\end{equation}

This ODE is a useful deterministic diagnostic before simulating the reverse SDE.


\section{Weak-form variational training}
\label{sec:weak}

Enforcing the transport equation \eqref{eq:pik_strong} pointwise constrains the
field derivative $\partial_\phi v_\theta$. In $d$ spatial dimensions the
analogous term is the divergence $\nabla\!\cdot v_\theta$, whose exact evaluation
costs $\mathcal{O}(d)$ backward passes and whose stochastic (Hutchinson) estimate
injects variance into the loss. We therefore adopt a weak, divergence-free
objective as the training method.

We represent the transport velocity as the gradient of a scalar potential,

\begin{equation}
 v_\theta(\phi,t)=\partial_\phi s_\theta(\phi,t).
 \label{eq:potential_velocity}
\end{equation}

The $\phi\mapsto-\phi$ symmetry is imposed by taking $s_\theta$ even, for example
$s_\theta(\phi,t)=g_\theta(\phi^2,t)$, which makes $v_\theta$ odd and enforces
$v_\theta(0,t)=0$ exactly.

The variational objective is

\begin{equation}
 \mathcal{L}_{\mathrm{weak}}(\theta)
 =
 \int_0^1\!\dd t\,
 \left\langle
 \tfrac12\bigl(\partial_\phi s_\theta\bigr)^2
 +
 s_\theta\bigl(\partial_t\hat S_t-\langle\partial_t\hat S_t\rangle_{p_t}\bigr)
 \right\rangle_{p_t}.
 \label{eq:weak_loss}
\end{equation}

Its first variation, integrated by parts under the zero-flux condition
\eqref{eq:zero_flux}, gives

\begin{equation}
 \partial_\phi\bigl(p_t\,\partial_\phi s_\theta\bigr)=-\partial_t p_t,
 \label{eq:weak_euler_lagrange}
\end{equation}

which is the continuity equation \eqref{eq:continuity} with
$v_t=\partial_\phi s_\theta$. Furthermore

\begin{equation}
 \mathcal{L}_{\mathrm{weak}}(\theta)-\mathcal{L}_{\mathrm{weak}}(\theta^\star)
 =
 \tfrac12\int_0^1\!\dd t\,
 \bigl\langle\bigl(\partial_\phi s_\theta-v_t^{\mathrm{exact}}\bigr)^2\bigr\rangle_{p_t}
 \ge 0,
 \label{eq:weak_lower_bound}
\end{equation}

so the objective is bounded below and its unique minimizer is the exact PIK
velocity \eqref{eq:exact_velocity_left}. Equation~\eqref{eq:weak_loss} is the
Action-Matching objective of Neklyudov \textit{et al.}\ specialized to an
analytically prescribed action path: because
$\partial_t\log p_t=-(\partial_t\hat S_t-\langle\partial_t\hat S_t\rangle_{p_t})$
is known in closed form, the time-boundary terms of the general construction
collapse to the single source term in \eqref{eq:weak_loss}.

Two properties make this objective numerically benign. It contains only the first
field derivative $\partial_\phi s_\theta$ and no divergence, so it scales to high
dimension at the cost of a single gradient $\nabla s_\theta$. And the gauge
direction $s_\theta\mapsto s_\theta+c(t)$ is automatically annihilated, because
$\langle\partial_t\hat S_t-\langle\partial_t\hat S_t\rangle_{p_t}\rangle_{p_t}=0$
and a $\phi$-independent shift has vanishing field gradient. The objective is
therefore coercive without an explicit boundary term; we optionally fix
$s_\theta(0,t)=0$ to make the potential unique.

\subsection{Discretization}

On the field grid $\{\phi_i\}$ with quadrature weights $w_i$ we use the discrete
probability weights

\begin{equation}
 \pi_i(t)=\frac{e^{-\hat S_t(\phi_i)}\,w_i}{\sum_j e^{-\hat S_t(\phi_j)}\,w_j},
 \label{eq:discrete_probability_weights}
\end{equation}

and approximate \eqref{eq:weak_loss} on $N_t$ time slices by

\begin{align}
 \mathcal{L}_{\mathrm{weak}}^{\mathrm{quad}}
 ={}&
 \frac{1}{N_t}\sum_{a=1}^{N_t}\sum_i \pi_i(t_a)
 \Bigl[
 \tfrac12\bigl(\partial_\phi s_\theta(\phi_i,t_a)\bigr)^2
 \nonumber\\
 &+
 s_\theta(\phi_i,t_a)\,\sigma_i(t_a)
 \Bigr],
 \label{eq:weak_quadrature_loss}
\end{align}

with
$\sigma_i(t_a)=\partial_t\hat S_{t_a}(\phi_i)-\sum_j\pi_j(t_a)\partial_t\hat S_{t_a}(\phi_j)$.
In contrast to the strong residual, which in higher dimensions requires the
divergence $\nabla\!\cdot v_\theta$, the weak objective needs only the gradient
$\nabla s_\theta$, whose cost is independent of dimension.

\subsection{Zero-dimensional benchmark}

For the target action \eqref{eq:target_action} and Gaussian base
\eqref{eq:base_action} with the smoothstep schedule \eqref{eq:schedule}, we
minimized \eqref{eq:weak_quadrature_loss} for an even four-layer multilayer
perceptron $s_\theta(\phi^2,t)$ on the full grid ($N=4097$ on $[-8,8]$, $N_t=128$,
double precision, Adam with cosine learning-rate decay).
Figure~\ref{fig:v_compare} compares the learned velocity
with the exact reference \eqref{eq:exact_velocity_left}: the probability-weighted
relative error is below $0.2\%$ across the bulk for $t\le0.7$ and about $1.3\%$
near the Gaussian endpoint, where the velocity itself is small. Integrating the
reverse probability-flow ODE \eqref{eq:reverse_probability_flow} from the base
distribution reproduces the target moments, $\langle\phi^2\rangle=3.408$ and
$\langle\phi^4\rangle=15.62$ against the quadrature values $3.409$ and $15.63$, and
the resulting trajectories track the exact cumulative-distribution quantile map
\eqref{eq:quantile_map}, as shown in Figure~\ref{fig:reverse_traj}.
